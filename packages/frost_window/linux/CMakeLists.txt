cmake_minimum_required(VERSION 3.13)
set(PROJECT_NAME "frost_window")
project(${PROJECT_NAME} LANGUAGES CXX)
set(PLUGIN_NAME "frost_window_plugin")
find_package(PkgConfig REQUIRED)
pkg_check_modules(FROST_GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
add_library(${PLUGIN_NAME} SHARED "frost_window_plugin.cc" "caption_appearance.cc" "window_geometry.cc")
apply_standard_settings(${PLUGIN_NAME})
set_target_properties(${PLUGIN_NAME} PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::FROST_GTK)
set(frost_window_bundled_libraries "" PARENT_SCOPE)
